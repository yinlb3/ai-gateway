# Gateway only
# --------------------------------------------------------------------------
# Take the whole LiteLLM set from pip with its extras, not from conda
# package by package. LiteLLM pins litellm-proxy-extras to an exact
# version per release, so pip is what keeps the pair in step.
#
# conda-forge is not a substitute here: its litellm recipe carries only
# the core SDK and no proxy dependency at all, no fastapi, no uvicorn,
# no litellm-proxy-extras, and it lagged at 1.82.6 while PyPI was at
# 1.102.0. Installing the pieces from different sources leaves a
# combination no one upstream has tested.
#
#     pip install "litellm[proxy,extra-proxy]"
#
# The proxy extra carries the web server, the Redis client and the
# virtual-key UI. extra-proxy adds the database and cloud-storage pieces,
# including prisma, which must also be on PATH as a command rather than
# only importable: LiteLLM calls subprocess.run(["prisma"]) and skips
# table creation when the lookup fails, after which every database query
# answers 502.

litellm[proxy,extra-proxy]   pip

# Pin prisma by hand. The extra declares prisma>=0.11.0,<1.0, so a plain
# install takes the newest release, but upstream CI only ever tests
# 0.11.0. With 0.15.0 the proxy dies at startup on
# `AttributeError: _Prisma__engine` (issue #39114, PR #39117, both open).
# After changing this pin, regenerate the client, or it reports
# `AttributeError: module 'prisma._types' has no attribute 'SortMode'`:
#
#     python -m prisma generate --schema <project root>/schema.prisma

prisma==0.11.0               pip
